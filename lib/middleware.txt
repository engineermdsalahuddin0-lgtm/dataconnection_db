CASH_PERCENT = 0.30
BANK_PERCENT = 0.70

function qualifies_for_cash(gross, DOJ):
    if gross <= 0:
        return false
    if DOJ < SEP_1:
        return gross >= 60000
    else:
        return gross >= 100000

function calculate_salary_split(emp):
    gross = emp.Gross

    # 1. threshold check
    if not qualifies_for_cash(gross, emp.DOJ):
        return {
            bank: { Basic: emp.Basic, HouseRent: emp.HouseRent,
                    Medical: emp.Medical, Conveyance: emp.Conveyance,
                    Total: gross },
            cash: null
        }

    # 2. 70 / 30 split of gross
    bank_total = round(gross * BANK_PERCENT)
    cash_total = gross - bank_total          # remainder, jate total mile

    # 3. employee's own component percentage (based on gross)
    basic_pct      = emp.Basic      / gross
    houserent_pct  = emp.HouseRent  / gross
    medical_pct    = emp.Medical    / gross
    conveyance_pct = emp.Conveyance / gross

    # 4. cash part breakdown
    cash_houserent  = round(cash_total * houserent_pct)
    cash_medical    = round(cash_total * medical_pct)
    cash_conveyance = round(cash_total * conveyance_pct)
    cash_basic      = cash_total - (cash_houserent + cash_medical + cash_conveyance)  # remainder

    # 5. bank part breakdown (same percentage)
    bank_houserent  = round(bank_total * houserent_pct)
    bank_medical    = round(bank_total * medical_pct)
    bank_conveyance = round(bank_total * conveyance_pct)
    bank_basic      = bank_total - (bank_houserent + bank_medical + bank_conveyance)

    return {
        bank: { Basic: bank_basic, HouseRent: bank_houserent,
                Medical: bank_medical, Conveyance: bank_conveyance,
                Total: bank_total },
        cash: { Basic: cash_basic, HouseRent: cash_houserent,
                Medical: cash_medical, Conveyance: cash_conveyance,
                Total: cash_total }
    }
